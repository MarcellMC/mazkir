# Fast Lane A2: the fast lane writes — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** With `FAST_LANE_MODE=on`, a tracking message is parsed once, placed on the timeline by code, matched to the ledger, and written through the existing tool executor. You get a receipt with ✓ ◌ ● ? lines, habit ticks, Google sync after the receipt, and a 24-hour undo. Everything else falls through to a skill chosen by the parse, with no router call.

**Architecture:** A1's parse and resolver are reused as they are. New modules under `apps/vault-server/src/services/fast_lane/` handle the rest:
- `matching.py` matches evidence to the ledger (§7.1);
- `habits.py` and `places.py` handle habits and places (§7.2, §7.3);
- `actions.py` turns clauses into planned tool calls;
- `act.py` runs them through `AgentService.run_tool` and keeps a per-turn action log with undo snapshots;
- `receipt.py` renders the receipt;
- `turn.py` orchestrates;
- `undo.py` reverses a turn.

The message route branches on `on`. Receipts and stages stream to the bot over SSE, and a new rich formatter draws them. `/day` shares the plan intent, the habit hook, the deferred Google sync and the photo placement.

**Tech Stack:**
- Python 3.14, FastAPI, pytest, `zoneinfo`, `rapidfuzz`, Anthropic SDK (Haiku, structured output; already wired by A1);
- TypeScript, grammY 1.46, Bot API rich messages, vitest.

**Spec:** `docs/specs/2026-09-27-fast-lane-design.md` (binding). A1's plan is `docs/plans/2026-09-27-fast-lane-a1.md`; its merged code is the base.

```python
def has_begun(event: dict[str, Any], now: dt.datetime) -> bool:
    """Whether the block's start — or its end, when it has no start — is at or before `now`.

    Local wall clock on both sides, the ledger's convention: an offset on
    either value is dropped, not converted.
    """
    value = event.get("start_time") or event.get("end_time")
    if not value:
        return False
    try:
        moment = dt.datetime.fromisoformat(str(value))
    except ValueError:
        return False
    return moment.replace(tzinfo=None) <= now.replace(tzinfo=None)


def _local_now() -> dt.datetime:
    return dt.datetime.now(ZoneInfo(settings.vault_timezone)).replace(tzinfo=None)
```

```python
def _wall(moment: dt.datetime) -> dt.datetime:
    return moment.replace(tzinfo=None)


def retract_completion(vault: Any, path: str, completed_at: str, prior: dict) -> dict:
    """Take back one completion that `complete_habit` recorded (fast-lane spec §9.3).

    Removes the log entry stamped `completed_at`, restores `streak`,
    `longest_streak` and `last_completed` from `prior` (what `complete_habit`
    returned), and takes the tokens back from the totals and from the note of
    the completion's own day. Refuses when a later completion exists:
    restoring the snapshot would erase that one's streak as well.
    """
    from src.services.daily_tasks import replace_or_append_section

    at = dt.datetime.fromisoformat(completed_at)
    habit = vault.read_file(path)
    meta, body = habit["metadata"], habit.get("content", "")
    entries = parse_completion_log(body)
    index = next((k for k, e in enumerate(entries) if _wall(e.at) == _wall(at)), None)
    if index is None:
        return {"ok": False, "reason": "completion not found"}
    if any(_wall(e.at) > _wall(at) for e in entries):
        return {"ok": False, "reason": "a later completion exists"}
    kept = entries[:index] + entries[index + 1:]
    new_body = replace_or_append_section(
        body, "Completion Log", "## Completion Log\n" + render_completion_log(kept),
    )
    vault.write_file(path, {
        **meta,
        "streak": prior.get("streak", meta.get("streak", 0)),
        "longest_streak": prior.get("longest_streak", meta.get("longest_streak", 0)),
        "last_completed": prior.get("last_completed"),
    }, new_body)
    tokens = meta.get("tokens_per_completion", 5)
    vault.update_tokens(-tokens, f"undo: {meta.get('name', 'habit')}", date=_wall(at).date().isoformat())
    return {"ok": True, "tokens_retracted": tokens}
```

```python
def _find_todo(lines: list[str], text: str) -> tuple[int | None, str]:
    """The one checkbox line whose text contains `text`, or (None, "not_found" | "ambiguous").

    Case-insensitive substring; a needle shorter than 3 characters is refused
    as "not_found" (see `set_todo_checked`). `moved` lines are skipped.
    """
    needle = text.strip().lower()
    if len(needle) < 3:
        return None, "not_found"
    hits = []
    for index, line in enumerate(lines):
        fields = _parse_todo_line(line)
        if fields is None or fields["state"] == "moved":
            continue
        if needle in fields["text"].strip().lower():
            hits.append(index)
    if not hits:
        return None, "not_found"
    if len(hits) > 1:
        return None, "ambiguous"
    return hits[0], "ok"


def remove_todo(body: str, text: str) -> tuple[str | None, str]:
    """Delete one checkbox line anywhere in the note, matched as `set_todo_checked` matches.

    Undo of an added todo (fast-lane spec §7.6). Only the matched line goes;
    every other line, in every section, is left byte-for-byte.
    """
    lines = body.splitlines(keepends=True)
    index, reason = _find_todo(lines, text)
    if index is None:
        return None, reason
    del lines[index]
    return "".join(lines), "ok"
```

```python
"""Tell Google Calendar about one of Mazkir's blocks, and say exactly what happened.

The decision tree `update_event` always used, in one place so the fast lane
can run it after its receipt is on screen (fast-lane spec §7.5):

- a block with no start can't be told to Google;
- a span across midnight is left alone (Google keeps one event where Mazkir
  keeps two fragments);
- only Mazkir's own calendar is ours to change;
- an inferred block (a checkbox, a habit, a visit) is never duplicated into
  Google;
- a proposal waits for your ✓.

The result is the `calendar_sync` shape every reporting rule already reads.
"""

from __future__ import annotations

import logging
from typing import Any

logger = logging.getLogger(__name__)


def _skip(reason: str) -> dict:
    return {"ok": False, "attempted": False, "reason": reason}


def sync_block(calendar: Any, events: Any, date_str: str, event_id: str) -> dict:
    """Create or update the Google entry for one block. Never raises."""
    from src.services.async_bridge import maybe_await
    from src.services.events_service import is_complete
    from src.services.interval import crosses_midnight

    stored = next((e for e in events.get_events(date_str) if e.get("id") == event_id), None)
    if stored is None:
        return _skip("not_found")
    source_ids = stored.get("source_ids") or {}
    calendar_id = source_ids.get("calendar_id")
    if stored.get("proposed") and stored.get("state") != "approved":
        return _skip("proposed")
    if not stored.get("start_time"):
        return _skip("no_start_time")
    if is_complete(stored) and crosses_midnight(stored["start_time"], stored["end_time"]):
        return _skip("crosses_midnight")
    if stored.get("source") == "photo":
        return _skip("not_applicable")
    if not calendar or not getattr(calendar, "is_initialized", False):
        return _skip("calendar_not_configured")
    if calendar_id and stored.get("calendar") not in (None, "Mazkir"):
        return _skip("not_in_mazkir_calendar")
    if not calendar_id and source_ids:
        return _skip("derived_from_source")
    try:
        if calendar_id:
            pushed = maybe_await(calendar.update_event(
                event_id=calendar_id,
                name=stored.get("name"),
                start_time=stored.get("start_time"),
                end_time=stored.get("end_time"),
            ))
            sync = {"ok": bool(pushed), "attempted": True, "event_id": calendar_id}
            if not pushed:
                sync["reason"] = "update_failed"
            return sync
        gcal_id = maybe_await(calendar.create_event(
            name=stored.get("name"),
            date=date_str,
            start_time=(stored.get("start_time") or "")[11:16],
            end_time=(stored.get("end_time") or "")[11:16] or None,
        ))
        sync = {"ok": bool(gcal_id), "attempted": True, "event_id": gcal_id}
        if gcal_id:
            # Membership is recorded by calendar_id alone: without it the next
            # edit would create a second Google entry.
            events.update_event(date=date_str, event_id=event_id,
                                updates={"source_ids": {**source_ids, "calendar_id": gcal_id}})
        else:
            sync["reason"] = "create_failed"
        return sync
    except Exception as e:
        logger.warning("Failed to sync block to Google Calendar: %s", e)
        return {"ok": False, "attempted": True, "reason": str(e)}
```

```python
"""Places you name, so a block can say where it happened (fast-lane spec §7.3).

`data/places.json`, gitignored:

    {"places": [{"id": "gym", "name": "Gym", "aliases": ["the gym"], "kind": "sport"}]}

Names only in A: coordinates and arrival triggers come later. A known place
puts `place_id` on the block; an unknown one is kept as `place_text`, and the
receipt offers to add it.
"""

from __future__ import annotations

import datetime as dt
import json
import re
from collections import Counter
from dataclasses import dataclass
from pathlib import Path

_LEADING = re.compile(r"^(?:at|the|in|to|в|на)\s+")


@dataclass(frozen=True)
class Place:
    id: str
    name: str
    aliases: tuple[str, ...] = ()
    kind: str | None = None


def _norm(text: str) -> str:
    text = re.sub(r"\s+", " ", text.strip().casefold())
    while (stripped := _LEADING.sub("", text)) != text:   # "at the gym" → "gym"
        text = stripped
    return text


def slug(name: str) -> str:
    return re.sub(r"[^\w]+", "-", name.strip().casefold()).strip("-") or "place"


def load_places(path: Path) -> list[Place]:
    if not Path(path).exists():
        return []
    raw = json.loads(Path(path).read_text(encoding="utf-8"))
    return [Place(str(p["id"]), str(p["name"]), tuple(str(a) for a in p.get("aliases") or ()), p.get("kind"))
            for p in raw.get("places", [])]


def save_places(path: Path, places: list[Place]) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = {"places": [{"id": p.id, "name": p.name, "aliases": list(p.aliases), "kind": p.kind}
                          for p in places]}
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
    tmp.replace(path)


def match_place(text: str | None, places: list[Place]) -> Place | None:
    """The place `text` names, by its name or an alias, ignoring case and "at the"."""
    if not text:
        return None
    wanted = _norm(text)
    for place in places:
        if wanted in {_norm(place.name), *(_norm(a) for a in place.aliases)}:
            return place
    return None


def _unique_id(name: str, taken: set[str]) -> str:
    base = candidate = slug(name)
    n = 2
    while candidate in taken:
        candidate, n = f"{base}-{n}", n + 1
    return candidate


def add_place(path: Path, name: str, kind: str | None = None) -> Place:
    """Add `name` as a place, or return the one it already names."""
    places = load_places(path)
    existing = match_place(name, places)
    if existing is not None:
        return existing
    place = Place(_unique_id(name, {p.id for p in places}), name.strip(), (), kind)
    save_places(path, [*places, place])
    return place


def bootstrap_places(events, today: dt.date, days: int = 120, min_visits: int = 2) -> list[Place]:
    """Candidate places: location names already in the ledger, on `min_visits` blocks or more."""
    counts: Counter[str] = Counter()
    shown: dict[str, str] = {}
    for k in range(days):
        for e in events.get_events((today - dt.timedelta(days=k)).isoformat()):
            name = str((e.get("location") or {}).get("name") or "").strip()
            if name:
                counts[_norm(name)] += 1
                shown.setdefault(_norm(name), name)
    out: list[Place] = []
    for key, n in counts.most_common():
        if n >= min_visits:
            out.append(Place(_unique_id(shown[key], {p.id for p in out}), shown[key]))
    return out
```

- A block the habit itself produced (`source_ids.habit_slug`) is never ticked again. The `/day` habit branch already ticked it.

```python
"""Which habit a block is, and ticking it at the block's own time (fast-lane spec §7.2).

The ledger showed why this exists: 12 of 16 dog walks and every gym block
were logged and never ticked, 165 tokens unpaid. A block that *is* a habit
now ticks it, once, stamped when the block ended, so a walk logged after
midnight counts for the evening it happened.
"""

from __future__ import annotations

import datetime as dt
import json
import re
from dataclasses import dataclass
from pathlib import Path
from zoneinfo import ZoneInfo

from rapidfuzz import fuzz

from src.config import settings
from src.services.habit_completion import complete_habit

FUZZY_FLOOR = 85.0
SUGGEST_MIN_DAYS = 3
SUGGEST_WINDOW_DAYS = 14
SHOWN_QUIET = dt.timedelta(days=30)      # spec: at most once a month per activity
DECLINED_QUIET = dt.timedelta(days=90)   # "not now" is remembered for a season, not forever
_COMPLETION_KEYS = ("path", "name", "completed_at", "prior", "completions_today",
                    "daily_target", "tokens_earned", "new_streak", "longest_streak")


@dataclass(frozen=True)
class HabitMatch:
    path: str
    name: str
    kind: str          # "exact" | "alias" | "fuzzy"


def _norm(text) -> str:
    return re.sub(r"\s+", " ", str(text or "").strip().casefold())


def match_habit(activity: str | None, tags, habits: list[dict]) -> HabitMatch | None:
    """The habit an activity is: by name, then by alias, then fuzzily (for a tap, never a tick)."""
    wanted = {_norm(activity), *(_norm(t) for t in tags or ())} - {""}
    if not wanted:
        return None
    for h in habits:
        meta = h.get("metadata", {})
        if _norm(meta.get("name")) in wanted:
            return HabitMatch(h["path"], str(meta.get("name", "")), "exact")
    for h in habits:
        meta = h.get("metadata", {})
        if {_norm(a) for a in meta.get("aliases") or ()} & wanted:
            return HabitMatch(h["path"], str(meta.get("name", "")), "alias")
    best, score = None, 0.0
    for h in habits:
        meta = h.get("metadata", {})
        for label in (meta.get("name"), *(meta.get("aliases") or ())):
            if activity and label:
                s = fuzz.token_set_ratio(_norm(activity), _norm(label))
                if s > score:
                    best, score = h, s
    if best is not None and score >= FUZZY_FLOOR:
        return HabitMatch(best["path"], str(best["metadata"].get("name", "")), "fuzzy")
    return None


def _event(events, date_str: str, event_id: str) -> dict | None:
    return next((e for e in events.get_events(date_str) if e.get("id") == event_id), None)


def tick_block_habit(vault, events, date_str: str, event_id: str, match: HabitMatch) -> dict | None:
    """Tick `match` for one block, stamped at the block's end, and record it on the block.

    A block ticks at most once: one already carrying `habit_completion` is
    left alone. Returns the completion recorded, or None when nothing was
    ticked (already ticked, or the day's target already met).
    """
    event = _event(events, date_str, event_id)
    if event is None or event.get("habit_completion"):
        return None
    stamp = event.get("end_time") or event.get("start_time")
    if not stamp:
        return None
    at = dt.datetime.fromisoformat(stamp).replace(tzinfo=ZoneInfo(settings.vault_timezone))
    outcome = complete_habit(vault, match.path, now=at)
    if outcome.get("already_completed"):
        return None
    completion = {k: outcome[k] for k in _COMPLETION_KEYS}
    events.update_event(date=date_str, event_id=event_id, updates={"habit_completion": completion})
    return completion


def tick_if_habit(vault, events, date_str: str, event_id: str, now: dt.datetime | None = None) -> dict | None:
    """The hook's rule: a confirmed block that has happened ticks its habit on an exact or alias match."""
    from src.services.approval import has_begun, resolve_state, source_systems

    now = now or dt.datetime.now(ZoneInfo(settings.vault_timezone)).replace(tzinfo=None)
    event = _event(events, date_str, event_id)
    if event is None or event.get("habit_completion") or "habit" in source_systems(event):
        return None
    confirmed = event.get("state") == "approved"
    if (event.get("proposed") or event.get("intent") == "plan") and not confirmed:
        return None
    if resolve_state(event, now) != "approved" or not has_begun(event, now):
        return None
    tags = [event["activity"]] if event.get("activity") else []
    match = match_habit(event.get("name"), tags, vault.list_active_habits())
    if match is None or match.kind == "fuzzy":
        return None
    return tick_block_habit(vault, events, date_str, event_id, match)


def _load(path: Path) -> dict:
    try:
        return json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return {}


def _save(path: Path, data: dict) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    Path(path).write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def _quiet(store: dict, key: str, today: dt.date) -> bool:
    for bucket, span in (("shown", SHOWN_QUIET), ("declined", DECLINED_QUIET)):
        when = store.get(bucket, {}).get(key)
        if when and today - dt.date.fromisoformat(when) < span:
            return True
    return False


def suggest_habit(name: str | None, events, habits: list[dict], today: dt.date, store_path: Path) -> str | None:
    """An activity on 3+ distinct days in 14 that is no habit yet: offer to track it, at most monthly."""
    if not name or match_habit(name, (), habits) is not None:
        return None
    key = _norm(name)
    days = {k for k in range(SUGGEST_WINDOW_DAYS)
            if any(_norm(e.get("name")) == key
                   for e in events.get_events((today - dt.timedelta(days=k)).isoformat()))}
    if len(days) < SUGGEST_MIN_DAYS:
        return None
    store = _load(store_path)
    if _quiet(store, key, today):
        return None
    store.setdefault("shown", {})[key] = today.isoformat()
    _save(store_path, store)
    return name


def decline_suggestion(name: str, today: dt.date, store_path: Path) -> None:
    store = _load(store_path)
    store.setdefault("declined", {})[_norm(name)] = today.isoformat()
    _save(store_path, store)
```

```python
from __future__ import annotations

import logging

logger = logging.getLogger(__name__)


def habit_hook(params: dict, output: dict, ctx) -> None:
    """Side effect only, and never raises: a failed tick must not fail the write."""
    try:
        if not isinstance(output, dict) or not output.get("ok"):
            return
        events, vault = ctx.get("events"), ctx.get("vault")
        if events is None or vault is None:
            return
        data = output.get("data") or {}
        first = data.get("event_id") or (data.get("event") or {}).get("id")
        if not first:
            return
        from src.services.fast_lane.habits import tick_if_habit
        # A block split at midnight ticks once, on its last fragment: that is when it ended.
        for event_id in reversed(list(dict.fromkeys([first, *(data.get("fragments") or [])]))):
            date = events.resolve_event_date(event_id, data.get("date") or params.get("date"))
            if date and tick_if_habit(vault, events, date, event_id):
                return
    except Exception:
        logger.warning("habit hook failed", exc_info=True)
```

```python
"""What a record does to the blocks already in the ledger (fast-lane spec §7.1).

Pure. The blocks come in as event dicts, each carrying the `date` of the file
it came from; the decision comes out as a `Match`. Nothing here writes: the
planner (`actions.py`) turns a `Match` into tool calls.
"""

from __future__ import annotations

import datetime as dt
import re
from dataclasses import dataclass
from typing import Any, Iterable

from src.services.block_resolver import resolve_block

SAME_PLAN_WINDOW = dt.timedelta(hours=2)
OVERLAP_SHARE = 0.5
SLEEP_WORDS = frozenset({"sleep", "sleeping", "nap", "сон", "שינה"})


@dataclass(frozen=True)
class Match:
    kind: str                     # create | close | confirm | update | also_close_other | sleep | move
    block: dict | None = None     # the ledger block it acts on


def _norm(text) -> str:
    return re.sub(r"\s+", " ", str(text or "").strip().casefold())


_ARTICLE = re.compile(r"^(?:the|my|that|this|a|an)\s+", re.IGNORECASE)


def _bare(reference: str) -> str:
    """"the gym" → "gym": block_resolver scores the article too, 40 for "the gym" against "Gym"."""
    return _ARTICLE.sub("", str(reference or "").strip())


def day_blocks(events, dates: Iterable[str]) -> list[dict[str, Any]]:
    """Every live block on `dates`, each tagged with the date of its file."""
    out = []
    for date in dict.fromkeys(d for d in dates if d):
        for e in events.get_events(date):
            if e.get("state") != "dismissed":
                out.append({**e, "date": date})
    return out


def instant(value: str | None, tz) -> dt.datetime | None:
    """A ledger timestamp (local wall clock) as an aware instant, or None."""
    if not value:
        return None
    try:
        moment = dt.datetime.fromisoformat(str(value))
    except ValueError:
        return None
    return moment.replace(tzinfo=tz) if moment.tzinfo is None else moment.astimezone(tz)


NAMED = 95   # block_resolver's score for an exact or contained name


def find_target(reference: str, *, reply_text: str | None, selected: list[dict],
                today: list[dict], later: list[dict] = (), prefer_open: bool = False) -> dict | None:
    """The block an edit is about: the replied-to message, the day open in /day, today, then the coming days.

    `prefer_open` (an end) tries each tier's open blocks first: "got back"
    closes the walk still running, not one that ended this morning. The
    coming days (`later`, e.g. a trip planned for the weekend) are reached
    only by an exact or contained name. None when no tier names exactly one
    block; the caller asks "which block?".
    """
    def pick(tier: list[dict], least: float = 0) -> dict | None:
        pools = [[b for b in tier if not b.get("end_time")], tier] if prefer_open else [tier]
        for pool in pools:
            if pool:
                found = resolve_block(_bare(reference), pool)
                if found.get("ok") and found["data"].get("score", 100) >= least:
                    return next(b for b in pool if b["id"] == found["data"]["id"])
        return None

    if reply_text:
        pool = list({b["id"]: b for b in [*selected, *today]}.values())
        named = [b for b in pool if b.get("name") and _norm(b["name"]) in _norm(reply_text)]
        if len(named) == 1:
            return named[0]
        if named and (block := pick(named)) is not None:
            return block
    for tier in (selected, today):
        if tier and (block := pick(tier)) is not None:
            return block
    return pick(list(later), NAMED) if later else None


def same_activity(name: str | None, tags, block: dict) -> bool:
    if name and resolve_block(_bare(name), [block]).get("ok"):
        return True
    activity = _norm(block.get("activity"))
    return bool(activity) and activity in {_norm(t) for t in tags or ()}


def _is_plan(block: dict) -> bool:
    """Intent rather than evidence: a plan, Mazkir's proposal, or a calendar entry nobody has confirmed."""
    if block.get("state") == "approved":
        return False
    return block.get("intent") == "plan" or block.get("proposed") is True or block.get("source") == "calendar"


def _is_sleep(block: dict) -> bool:
    return _norm(block.get("name")) in SLEEP_WORDS or _norm(block.get("activity")) in SLEEP_WORDS


def _share(block: dict, start, end, tz) -> float:
    """How much of the shorter interval the two share, 0..1; 0 when either is open."""
    bs, be = instant(block.get("start_time"), tz), instant(block.get("end_time"), tz)
    if None in (bs, be, start, end):
        return 0.0
    overlap = (min(be, end) - max(bs, start)).total_seconds()
    shortest = min((be - bs).total_seconds(), (end - start).total_seconds())
    return overlap / shortest if overlap > 0 and shortest > 0 else 0.0


def match_record(name: str | None, tags, start: dt.datetime | None, end: dt.datetime | None,
                 candidates: list[dict], tz, intent: str = "record") -> Match:
    """Spec §7.1, in order: close, confirm, update, Sleep, close-the-other, create.

    A plan only moves a plan or proposal of its activity within ±2 h, so it
    never adds a second one (revised 2026-10-03); it touches no record.
    """
    blocks = [b for b in candidates if b.get("state") != "dismissed"]
    same = [b for b in blocks if same_activity(name, tags, b)]
    if intent == "plan":
        if start is not None:
            for b in same:
                bs = instant(b.get("start_time"), tz)
                if _is_plan(b) and bs is not None and abs(bs - start) <= SAME_PLAN_WINDOW:
                    return Match("move", b)
        return Match("create")
    moment = start or end
    if end is not None:
        for b in same:
            bs = instant(b.get("start_time"), tz)
            if b.get("end_time") is None and bs is not None and bs <= moment:
                return Match("close", b)
    if start is not None:
        for b in same:
            bs = instant(b.get("start_time"), tz)
            if _is_plan(b) and bs is not None and abs(bs - start) <= SAME_PLAN_WINDOW:
                return Match("confirm", b)
    for b in same:
        if not _is_plan(b) and _share(b, start, end, tz) >= OVERLAP_SHARE:
            return Match("update", b)
    for b in blocks:
        if b not in same and _is_sleep(b) and _share(b, start, end, tz) > 0:
            return Match("sleep", b)
    if start is not None:
        for b in blocks:
            bs = instant(b.get("start_time"), tz)
            if b not in same and b.get("end_time") is None and bs is not None and bs < start:
                return Match("also_close_other", b)
    return Match("create")
```

```python
"""What you see after the fast lane acts: one line per thing it did (fast-lane spec §8.2).

Rendered from the turn record `act.py` keeps (`data/actions/{turn_id}.json`),
so a tap that changes one line re-renders the same receipt from the same
data. Every word, glyph and button is decided here; the bot's formatter only
draws the structure.
"""

from __future__ import annotations

import datetime as dt
from typing import Any

GLYPHS = {"fact": "✓", "plan": "◌", "proposal": "●", "question": "?"}
STATUS_GLYPHS = {"failed": "✗", "undone": "↶", "dismissed": "✕", "approved": "✓"}
UNDO_WINDOW = dt.timedelta(hours=24)
_LIVE = ("done", "approved", "pending")


def _naive(moment: dt.datetime) -> dt.datetime:
    return moment.replace(tzinfo=None)


def day_label(day: dt.date, today: dt.date) -> str:
    """"" today, then "yest", "tmrw", a weekday within the week, else "7 Sep"."""
    if day == today:
        return ""
    if day == today - dt.timedelta(days=1):
        return "yest"
    if day == today + dt.timedelta(days=1):
        return "tmrw"
    if abs((day - today).days) < 7:
        return day.strftime("%a")
    return f"{day.day} {day.strftime('%b')}"


def when_label(start: str | None, end: str | None, now: dt.datetime, end_precision: str | None = None) -> str:
    """"yest 23:15–23:35"; "09:00–~09:30" for a soft end; "00:10→" for a block still running."""
    s = dt.datetime.fromisoformat(start) if start else None
    e = dt.datetime.fromisoformat(end) if end else None
    first = s or e
    if first is None:
        return ""
    soft = "~" if end_precision in ("approx", "assumed") else ""
    if s and e:
        span = f"{s:%H:%M}–{soft}{e:%H:%M}"
    elif s:
        span = f"{s:%H:%M}→"
    else:
        span = f"…–{soft}{e:%H:%M}"
    return f"{day_label(first.date(), now.date())} {span}".strip()


def _sync_label(sync: dict | None) -> str | None:
    if not sync:
        return None
    if sync.get("reason") == "deferred":
        return "⏳ calendar"
    if sync.get("ok"):
        return "✓ calendar"
    if sync.get("attempted"):
        return f"⚠ calendar: {sync.get('reason') or 'failed'}"
    return None


def _habit_detail(habit: dict | None) -> str | None:
    if not habit:
        return None
    text = (f"🎯 {habit['name']} {habit['completions_today']}/{habit['daily_target']} · "
            f"+{habit['tokens_earned']} 🪙 · streak {habit['new_streak']}")
    before = (habit.get("prior") or {}).get("longest_streak")
    if before is not None and habit["new_streak"] > before:
        text += " · longest ever"
    return text


def _buttons(tid: str, line: dict) -> list[dict[str, str]]:
    def tap(label: str, action: str) -> dict[str, str]:
        return {"label": label, "data": f"rcpt:{tid}:{line['id']}:{action}"}

    status, out = line["status"], []
    if line["kind"] == "suggest":
        return [tap("Track it", "sy"), tap("Not now", "sn")] if status == "pending" else []
    if status == "pending":
        out += [tap("✓", "ok"), tap("✕", "no")]
    if line.get("event_id") and status in _LIVE and line["kind"] in ("block", "edit"):
        out.append({"label": "✎", "data": f"block:edit:{line['date']}:{line['event_id']}"})
    if line.get("habit_candidate") and not line.get("habit") and status in ("done", "approved"):
        out.append(tap(f"{line['habit_candidate']['name']}?", "hb"))
    if line.get("place_text") and status in _LIVE:
        out.append(tap("+ place", "pl"))
    return out


def _line(tid: str, line: dict, now: dt.datetime) -> dict[str, Any]:
    glyph = STATUS_GLYPHS.get(line["status"]) or GLYPHS.get(line["outcome"], "·")
    when = when_label(line.get("start"), line.get("end"), now, line.get("end_precision"))
    if not when and line.get("day"):
        when = day_label(dt.date.fromisoformat(line["day"]), now.date())
    return {
        "id": line["id"], "glyph": glyph, "when": when, "text": line["text"],
        "detail": _habit_detail(line.get("habit")), "sync": _sync_label(line.get("calendar_sync")),
        "reason": line.get("undo_note") or line.get("reason") or "", "status": line["status"],
        "buttons": _buttons(tid, line),
    }


def receipt_text(receipt: dict) -> str:
    """The receipt as plain lines: the non-rich fallback, and what the conversation log keeps."""
    rows = []
    for line in receipt["lines"]:
        head = " ".join(p for p in (line["glyph"], line["when"], line["text"]) if p)
        if line.get("reason"):
            head += f" — {line['reason']}"
        rows.append(head)
        rows += [f"  {sub}" for sub in (line.get("detail"), line.get("sync")) if sub]
    return "\n".join(rows)


def _ask(line: dict) -> str:
    """What a question line asks, armed as the bot's open question so a plain reply answers it."""
    reason = line.get("reason") or "no time given"
    return f"{line['text']} · {reason}" + (". Reply with one." if reason == "no time given" else "")


def render_receipt(turn: dict, now: dt.datetime) -> dict:
    tid = turn["turn_id"]
    lines = [_line(tid, line, now) for line in turn["lines"]]
    fresh = _naive(now) - _naive(dt.datetime.fromisoformat(turn["created_at"])) < UNDO_WINDOW
    undoable = fresh and any(line.get("undo") and line["status"] in _LIVE for line in turn["lines"])
    asking = next((line for line in turn["lines"] if line["status"] == "question"), None)
    receipt = {
        "turn_id": tid, "lines": lines,
        "undo": f"rcpt:{tid}:u:un" if undoable else None,
        "open_question": _ask(asking) if asking else None,
    }
    receipt["text"] = receipt_text(receipt)
    return receipt


def ran_summary(turn: dict, now: dt.datetime) -> list[str]:
    """What the fast lane already did, told to the skill that takes the rest (spec §5.5)."""
    out = []
    for line, drawn in zip(turn["lines"], render_receipt(turn, now)["lines"]):
        if line["status"] in _LIVE:
            out.append(" ".join(p for p in (drawn["glyph"], drawn["when"], drawn["text"]) if p)
                       + f" ({line['kind']})")
    return out

```

```python
"""Turn a message's clauses into the tool calls that carry them out (fast-lane spec §7).

Pure: the parse's clauses, the resolver's placements, the day's blocks,
habits and places go in; `PlannedLine`s come out. A line names the calls to
run now, the ones to run if you tap ✓, or the question it asks.
Nothing here writes, and nothing the model said becomes a timestamp or an
id: times are the resolver's and ids are the ledger's (spec §1.3).
"""

from __future__ import annotations

import datetime as dt
import re
from dataclasses import dataclass, field, replace
from typing import Any

from rapidfuzz import fuzz

from src.services.fast_lane.contract import ACTIVITY_TAGS, FAST_OPS, Clause, ParseResult
from src.services.fast_lane.habits import match_habit
from src.services.fast_lane.matching import instant, match_record
from src.services.fast_lane.places import Place, match_place
from src.services.fast_lane.time_resolver import (
    BLOCK_OPS, EDIT_OPS, ClauseResolution, Placement, to_wire,
)

STATED = frozenset({"exact", "approx"})   # ends the message wrote, not ones the resolver filled in
TODO_MATCH = 85                            # token_sort_ratio for the todo hook (spec §7.3)


@dataclass(frozen=True)
class Step:
    tool: str
    params: dict[str, Any]


@dataclass(frozen=True)
class PlannedLine:
    kind: str                     # block | edit | todo | habit | rollover | propose | question
    outcome: str                  # fact | plan | proposal | question
    text: str
    steps: tuple[Step, ...] = ()                    # run now
    pending: tuple[Step, ...] = ()                  # run on ✓
    placement: Placement | None = None
    reason: str = ""
    habit_candidate: dict | None = None             # a fuzzy habit match, confirmed by a tap
    place_text: str | None = None                   # a place not in places.json yet


@dataclass(frozen=True)
class PlanInputs:
    now: dt.datetime
    message_id: int | None
    candidates: list[dict] = field(default_factory=list)   # blocks on the days in play, each with "date"
    targets: dict[int, dict] = field(default_factory=dict)  # clause index → the block an edit is about
    habits: list[dict] = field(default_factory=list)
    places: list[Place] = field(default_factory=list)
    todos: list[str] = field(default_factory=list)   # today's open todos; a scheduled one starts with its time


def _wire(moment: dt.datetime | None) -> str | None:
    return to_wire(moment) if moment is not None else None


def _fields(c: Clause, p: Placement, inputs: PlanInputs, place: Place | None) -> dict[str, Any]:
    return {
        "intent": c.intent,
        "start_precision": p.start_precision, "end_precision": p.end_precision,
        "logical_date": p.logical_date.isoformat() if p.logical_date else None,
        "place_id": place.id if place else None,
        "place_text": c.place if c.place and place is None else None,
        "project": c.project,
        "evidence": {"text": c.evidence, "message_id": inputs.message_id},
        "tags": list(c.tags) or None,   # #green is the owner's reserved tag, kept for his analysis
    }


def _create(c: Clause, p: Placement, inputs: PlanInputs, *, proposed: bool = False) -> Step:
    place = match_place(c.place, inputs.places)
    start, end = _wire(p.start), _wire(p.end)
    params: dict[str, Any] = {
        "name": c.name or c.evidence, "date": (start or end)[:10],
        "_fast": {"defer_calendar_sync": True, "fields": _fields(c, p, inputs, place)},
    }
    if start:
        params["start_time"] = start
    if end:
        params["end_time"] = end
    if proposed:
        params["proposed"] = True
    activity = next((t for t in c.tags if t in ACTIVITY_TAGS), None)
    if activity:
        params["activity"] = activity
    links = [*c.people, *([c.project] if c.project else [])]
    if links:
        params["wikilinks"] = links
    if place:
        params["location"] = {"name": place.name}
    return Step("create_event", params)


def _update(block: dict, *, start: str | None = None, end: str | None = None, name: str | None = None,
            fields: dict | None = None, state: str | None = None) -> Step:
    """An `update_event` carrying only what differs from the block as it is now."""
    params: dict[str, Any] = {"event_id": block["id"], "date": block["date"],
                              "_fast": {"defer_calendar_sync": True, "fields": fields or {}}}
    if start and start != block.get("start_time"):
        params["start_time"] = start
    if end and end != block.get("end_time"):
        params["end_time"] = end
    if name and name != block.get("name"):
        params["name"] = name
    if state:
        params["_fast"]["state"] = state
    return Step("update_event", params)


def _open_todo(name: str, todos: list[str]) -> str | None:
    """The open todo a record of `name` finishes, when one clearly does (spec §7.3, the todo hook).

    token_sort_ratio, not token_set_ratio: "Gym" is a subset of "Renew the
    gym membership", which token_set_ratio scores 100.
    """
    best, best_score = None, 0.0
    for todo in todos:
        text = re.sub(r"^\d{1,2}:\d{2}\s+", "", todo)
        score = fuzz.token_sort_ratio(name.casefold(), text.casefold())
        if score > best_score:
            best, best_score = text, score
    return best if best_score >= TODO_MATCH else None


def _crossing_out(line: PlannedLine, name: str, inputs: PlanInputs) -> PlannedLine:
    """A record that finishes an open todo crosses it out in the same line, so undo reverses both."""
    todo = _open_todo(name, inputs.todos)
    if todo is None:
        return line
    step = Step("daily_check_todo", {"text": todo, "date": inputs.now.date().isoformat()})
    return replace(line, steps=line.steps + (step,))


def _block(c: Clause, r: ClauseResolution, inputs: PlanInputs) -> list[PlannedLine]:
    name = c.name or c.evidence
    if r.outcome == "question":
        return [PlannedLine("question", "question", name, reason=r.reason or "no time given")]
    p = r.placement
    known_place = match_place(c.place, inputs.places)
    common: dict[str, Any] = {"placement": p, "place_text": c.place if c.place and known_place is None else None}
    habit = match_habit(name, c.tags, inputs.habits)
    candidate = {"path": habit.path, "name": habit.name} if habit and habit.kind == "fuzzy" else None
    if r.outcome == "proposal":
        return [PlannedLine("block", "proposal", name, steps=(_create(c, p, inputs, proposed=True),),
                            reason=r.reason, **common)]
    tz = inputs.now.tzinfo
    precision = {"start_precision": p.start_precision, "end_precision": p.end_precision}
    if r.outcome == "plan":
        # A plan moves a plan or proposal of its activity within ±2 h rather than
        # adding a second (§7.1, revised 2026-10-03); a proposal it moves is approved.
        m = match_record(name, c.tags, p.start, p.end, inputs.candidates, tz, intent="plan")
        if m.kind == "move":
            step = _update(m.block, start=_wire(p.start), end=_wire(p.end),
                           fields={"intent": "plan", **precision},
                           state="approved" if m.block.get("proposed") else None)
            return [PlannedLine("edit", "plan", m.block.get("name") or name, steps=(step,), **common)]
        return [PlannedLine("block", "plan", name, steps=(_create(c, p, inputs),), **common)]
    m = match_record(name, c.tags, p.start, p.end, inputs.candidates, tz)
    if m.kind == "close":
        opened = instant(m.block.get("start_time"), tz)
        earlier = p.start is not None and opened is not None and p.start < opened
        step = _update(m.block, start=_wire(p.start) if earlier else None, end=_wire(p.end),
                       fields=precision if earlier else {"end_precision": p.end_precision})
        return [_crossing_out(PlannedLine("edit", "fact", m.block.get("name") or name, steps=(step,),
                                          habit_candidate=candidate, **common), name, inputs)]
    if m.kind == "confirm":
        step = _update(m.block, start=_wire(p.start), end=_wire(p.end),
                       fields={"intent": "record", **precision}, state="approved")
        return [_crossing_out(PlannedLine("edit", "fact", m.block.get("name") or name, steps=(step,),
                                          habit_candidate=candidate, **common), name, inputs)]
    if m.kind == "update":
        # Only the ends this record states: an assumed or inferred end never
        # overwrites one the block already has (§7.1, revised 2026-10-03).
        start = _wire(p.start) if p.start_precision in STATED or not m.block.get("start_time") else None
        end = _wire(p.end) if p.end_precision in STATED or not m.block.get("end_time") else None
        fields = {k: v for k, v, kept in (("start_precision", p.start_precision, start),
                                          ("end_precision", p.end_precision, end)) if kept}
        step = _update(m.block, start=start, end=end, fields=fields)
        return [_crossing_out(PlannedLine("edit", "fact", m.block.get("name") or name, steps=(step,),
                                          habit_candidate=candidate, **common), name, inputs)]
    if m.kind == "sleep":
        return [PlannedLine("block", "proposal", name, steps=(_create(c, p, inputs, proposed=True),),
                            reason=f"overlaps {m.block.get('name') or 'Sleep'}", **common)]
    lines = [_crossing_out(PlannedLine("block", "fact", name, steps=(_create(c, p, inputs),),
                                       habit_candidate=candidate, **common), name, inputs)]
    if m.kind == "also_close_other" and p.start is not None:
        at = _wire(p.start)
        lines.append(PlannedLine(
            "propose", "proposal", f"End {m.block.get('name')} at {at[11:16]}?",
            pending=(_update(m.block, end=at, fields={"end_precision": "inferred"}),), reason="still open",
        ))
    return lines


def _edit(i: int, c: Clause, r: ClauseResolution, inputs: PlanInputs) -> list[PlannedLine]:
    label = c.target or c.name or c.evidence
    target = inputs.targets.get(i)
    if target is None and c.op == "end_block" and r.placement is not None:
        # Nothing open to end: the resolver placed it as a new block ending then (§6.2 rule 9).
        return _block(replace(c, op="log_block", name=c.name or c.target), r, inputs)
    if target is None:
        return [PlannedLine("question", "question", label, reason="which block?")]
    if r.outcome == "question":
        return [PlannedLine("question", "question", target.get("name") or label, reason=r.reason)]
    p = r.placement
    fields = {}
    if p is not None:
        fields = {k: v for k, v in (("start_precision", p.start_precision), ("end_precision", p.end_precision)) if v}
    new_name = c.name if c.op == "edit_block" and c.name else None
    step = _update(target, start=_wire(p.start) if p else None, end=_wire(p.end) if p else None,
                   name=new_name, fields=fields)
    text = new_name or target.get("name") or label
    if r.outcome == "proposal":
        return [PlannedLine("propose", "proposal", text, pending=(step,), placement=p, reason=r.reason)]
    return [PlannedLine("edit", "fact", text, steps=(step,), placement=p)]


def _habit(c: Clause, r: ClauseResolution, inputs: PlanInputs) -> list[PlannedLine]:
    name = c.name or c.evidence
    match = match_habit(name, c.tags, inputs.habits)
    shown = match.name if match else name
    step = Step("complete_habit", {"habit_name": shown})
    if r.outcome == "proposal":
        return [PlannedLine("propose", "proposal", shown, pending=(step,), reason=r.reason or "mentioned in passing")]
    return [PlannedLine("habit", "fact", shown, steps=(step,))]


def _todo_text(c: Clause, inputs: PlanInputs) -> str:
    text = c.name or c.evidence
    # extract_hashtags yields "#dev-session" as both "dev-session" and "dev": keep the full tag once.
    tags = [t for t in dict.fromkeys(c.tags)
            if not any(o != t and (o.startswith(t + "-") or o.startswith(t + "/")) for o in c.tags)]
    if tags:
        text += " " + " ".join(f"#{t}" for t in tags)
    if c.place:
        place = match_place(c.place, inputs.places)
        text += f" @ {place.name if place else c.place}"
    return text


def _todo(c: Clause, r: ClauseResolution, inputs: PlanInputs) -> list[PlannedLine]:
    p = r.placement
    if c.op == "rollover_todos":
        # To the clause's day (today by default), from today when that lies ahead,
        # else from yesterday; a target moves only that todo (§7.3, revised).
        today = inputs.now.date()
        to = p.logical_date if p is not None and p.logical_date else today
        source = today if to > today else today - dt.timedelta(days=1)
        params = {"from_date": source.isoformat(), "to_date": to.isoformat()}
        if c.target:
            params["text"] = c.target
        text = f"Moved “{c.target}” to {to.isoformat()}" if c.target else f"Rolled over {source.isoformat()}'s todos"
        return [PlannedLine("rollover", "fact", text, steps=(Step("daily_rollover", params),))]
    day = (p.logical_date if p is not None and p.logical_date else inputs.now.date()).isoformat()
    if c.op == "check_todo":
        text = c.target or c.name or c.evidence
        step = Step("daily_check_todo", {"text": text, "date": day})
    else:
        text = _todo_text(c, inputs)
        step = Step("daily_add_task", {"text": text, "date": day})
    if r.outcome == "proposal":
        return [PlannedLine("propose", "proposal", text, pending=(step,), placement=p,
                            reason=r.reason or "mentioned in passing")]
    return [PlannedLine("todo", "fact", text, steps=(step,), placement=p)]


def plan_turn(result: ParseResult, resolutions: list[ClauseResolution | None],
              inputs: PlanInputs) -> list[PlannedLine]:
    """One or more lines per fast clause, in message order; `other` clauses are left to the skill."""
    lines: list[PlannedLine] = []
    for i, (c, r) in enumerate(zip(result.clauses, resolutions)):
        if c.op not in FAST_OPS or r is None:
            continue
        if c.op in BLOCK_OPS:
            lines += _block(c, r, inputs)
        elif c.op in EDIT_OPS:
            lines += _edit(i, c, r, inputs)
        elif c.op == "tick_habit":
            lines += _habit(c, r, inputs)
        else:
            lines += _todo(c, r, inputs)
    return lines
```

```python
def resolve_clauses(result: ParseResult, ctx: FastContext, settings: ShadowSettings,
                    targets: dict[int, dict] | None = None) -> list[ClauseResolution | None]:
    """The resolver's answer for each fast clause; None for clauses that fall through.

    `targets` (A2) maps an edit clause's index to the ledger block it is
    about: the replied-to message, the /day day or today (`matching.find_target`).
    Without it, A1's lookup among the snapshot's blocks is used.
    """
    candidates = [b.as_candidate() for b in ctx.blocks]
    by_id = {b.id: b for b in ctx.blocks}
    times = []
    for i, c in enumerate(result.clauses):
        time_ = _clause_time(c, candidates, by_id)
        target = (targets or {}).get(i)
        if target is not None:
            time_ = replace(time_, target_start=instant(target.get("start_time"), ctx.now.tzinfo),
                            target_end=instant(target.get("end_time"), ctx.now.tzinfo))
        times.append(time_)
    rctx = ResolverContext(now=ctx.now, day_boundary_hour=settings.day_boundary_hour,
                           typical_minutes=dict(ctx.typical_minutes), default_minutes=settings.default_minutes,
                           bedtime=ctx.bedtime, photo_at=ctx.photo_at)
    resolved = resolve(times, rctx)
    return [r if c.op in FAST_OPS else None for c, r in zip(result.clauses, resolved)]
```

```python
"""Carry out a planned turn, one call at a time, keeping what undo needs (fast-lane spec §7.5–7.6).

Every call goes through `AgentService.run_tool`, so the audit log, pinning,
hooks and cross-date moves behave exactly as they do for the agent. Calls
run in message order. A failing call marks its line ✗ with the reason, and
the other lines still run: nothing is claimed that did not happen.

The turn record lands in `data/actions/{turn_id}.json`. The receipt is
rendered from it, undo reverses it for 24 hours, and a repeat of the same
Telegram message within 10 minutes finds it instead of writing twice.
"""

from __future__ import annotations

import copy
import datetime as dt
import json
import re
import time
from dataclasses import asdict
from pathlib import Path
from typing import Any, Callable
from uuid import uuid4

from opentelemetry import trace

from src.services.fast_lane.actions import PlannedLine, Step
from src.services.fast_lane.receipt import UNDO_WINDOW
from src.services.fast_lane.time_resolver import to_wire

REPEAT_WINDOW = dt.timedelta(minutes=10)
_TURN_ID = re.compile(r"^t[0-9a-f]{8}$")
_HABIT_KEYS = ("path", "completed_at", "prior", "completions_today", "daily_target",
               "tokens_earned", "new_streak", "longest_streak")
_tracer = trace.get_tracer("mazkir.fast_lane")

RunTool = Callable[[str, dict], dict]


def new_turn_id() -> str:
    return "t" + uuid4().hex[:8]


def _find(events, event_id: str, date: str | None = None) -> dict | None:
    """A copy of the stored event, with the date of its file, or None."""
    found_date = events.resolve_event_date(event_id, date)
    if found_date is None:
        return None
    event = next((e for e in events.get_events(found_date) if e.get("id") == event_id), None)
    return {**copy.deepcopy(event), "date": found_date} if event else None


def _error(result: Any) -> str:
    error = result.get("error") if isinstance(result, dict) else None
    if isinstance(error, dict):
        return str(error.get("message") or error.get("code") or "failed")
    return str(error or "failed")


def run_step(step: Step, *, run_tool: RunTool, events) -> dict:
    """One tool call, and what the receipt and undo need to know about it."""
    before = None
    if step.tool == "update_event":
        before = _find(events, step.params["event_id"], step.params.get("date"))
    with _tracer.start_as_current_span(f"fast.act.{step.tool}"):
        result = run_tool(step.tool, step.params)
    if not isinstance(result, dict) or not result.get("ok"):
        return {"ok": False, "reason": _error(result), "undo": []}
    data = result.get("data") or {}
    out: dict[str, Any] = {"ok": True, "reason": "", "undo": []}
    if step.tool == "create_event":
        ids = list(dict.fromkeys([data["event_id"], *(data.get("fragments") or [])]))
        file_date = Path(data["path"]).stem
        after = {eid: _find(events, eid, file_date) for eid in ids}
        first = after[ids[0]] or {}
        out.update(event_id=ids[0], date=first.get("date", file_date), calendar_sync=data.get("calendar_sync"),
                   habit=next((e["habit_completion"] for e in after.values() if e and e.get("habit_completion")), None),
                   undo=[{"kind": "created", "ids": ids, "after": after}])
    elif step.tool == "update_event":
        event_id = step.params["event_id"]
        after = _find(events, event_id, data.get("date"))
        newly_ticked = (after or {}).get("habit_completion") if not (before or {}).get("habit_completion") else None
        out.update(event_id=event_id, date=(after or {}).get("date") or data.get("date"),
                   calendar_sync=data.get("calendar_sync"), habit=newly_ticked,
                   undo=[{"kind": "edited", "event_id": event_id, "before": before, "after": after}])
    elif step.tool == "complete_habit":
        out.update(habit={"name": data.get("habit"), **{k: data.get(k) for k in _HABIT_KEYS}},
                   undo=[{"kind": "habit", "path": data.get("path"), "completed_at": data.get("completed_at"),
                          "prior": data.get("prior")}])
    elif step.tool in ("daily_add_task", "daily_check_todo"):
        kind = "todo_added" if step.tool == "daily_add_task" else "todo_checked"
        out.update(day=data.get("date"), undo=[{"kind": kind, "date": data.get("date"), "text": data.get("text")}])
    elif step.tool == "daily_rollover":
        out.update(undo=[{"kind": "none", "why": "a rollover can't be undone from here"}])
    return out


def _record(k: int, line: PlannedLine) -> dict[str, Any]:
    p = line.placement
    return {
        "id": k, "kind": line.kind, "outcome": line.outcome, "text": line.text, "status": "", "reason": line.reason,
        "start": to_wire(p.start) if p and p.start else None,
        "end": to_wire(p.end) if p and p.end else None,
        "end_precision": p.end_precision if p else None,
        "day": p.logical_date.isoformat() if p and p.logical_date and not (p.start or p.end) else None,
        "date": None, "event_id": None, "calendar_sync": None, "habit": None,
        "habit_candidate": line.habit_candidate, "place_text": line.place_text,
        "pending": [asdict(s) for s in line.pending], "undo": [],
    }


def apply_result(rec: dict, res: dict, *, done_status: str) -> None:
    """Fold one call's outcome into its line. Shared with the receipt routes (Task 16)."""
    if not res["ok"]:
        rec["status"], rec["reason"] = "failed", res["reason"]
        return
    for key in ("event_id", "date", "calendar_sync", "habit", "day"):
        if res.get(key) is not None:
            rec[key] = res[key]
    rec["undo"] = [*rec.get("undo", []), *res["undo"]]
    rec["status"] = done_status


def run_lines(lines: list[PlannedLine], *, run_tool: RunTool, events) -> list[dict]:
    records = []
    for k, line in enumerate(lines):
        rec = _record(k, line)
        if line.kind == "question":
            rec["status"] = "question"
        elif line.kind == "propose":
            rec["status"] = "pending"
        else:
            done = "pending" if line.outcome == "proposal" else "done"
            for step in line.steps:
                res = run_step(step, run_tool=run_tool, events=events)
                apply_result(rec, res, done_status=done)
                if not res["ok"]:
                    break
        records.append(rec)
    return records


def _path(actions_dir: Path, turn_id: str) -> Path | None:
    return Path(actions_dir) / f"{turn_id}.json" if _TURN_ID.match(turn_id or "") else None


def save_turn(actions_dir: Path, turn: dict) -> None:
    path = _path(actions_dir, turn["turn_id"])
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(turn, ensure_ascii=False, default=str), encoding="utf-8")
    tmp.replace(path)


def load_turn(actions_dir: Path, turn_id: str) -> dict | None:
    """The turn record, or None. Only a `t<8 hex>` id is ever turned into a path."""
    path = _path(actions_dir, turn_id)
    if path is None or not path.exists():
        return None
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return None


def find_repeat(actions_dir: Path, chat_id: int, message_id: int | None) -> dict | None:
    """The turn this Telegram message already produced within 10 minutes (spec §10), or None."""
    if message_id is None or not Path(actions_dir).exists():
        return None
    horizon = time.time() - REPEAT_WINDOW.total_seconds()
    for path in Path(actions_dir).glob("t*.json"):
        if path.stat().st_mtime < horizon:
            continue
        turn = load_turn(actions_dir, path.stem)
        if turn and turn.get("chat_id") == chat_id and turn.get("message_id") == message_id:
            return turn
    return None


def prune(actions_dir: Path) -> None:
    """Undo snapshots are kept for 24 hours (spec §9.4)."""
    horizon = time.time() - UNDO_WINDOW.total_seconds()
    for path in Path(actions_dir).glob("t*.json"):
        try:
            if path.stat().st_mtime < horizon:
                path.unlink()
        except OSError:
            continue
```

Spec §7.6 sets the rules:
- Undo reverses a whole turn, from snapshots.
- A line is refused if its item changed since.
- Undo is available for 24 hours.

What each kind of line becomes:
- **A created block:** deleted, together with its Google entry.
- **An edited block:** its prior fields restored.
- **An added todo:** removed.
- **A checked todo:** unchecked.
- **A habit tick:** taken back (§9.3).

A tick the habit hook made on a block is taken back along with the block. Lines are undone newest first, so a later line never depends on an earlier one that is already gone.

```python
"""Take a fast-lane turn back, line by line (fast-lane spec §7.6, §9.3).

From the snapshots `act.py` kept at write time. A line whose item changed
since — you fixed the block in /day, say — is refused with the reason rather
than overwritten: undo restores what Mazkir did, never what you did after.
Available for 24 hours.
"""

from __future__ import annotations

import datetime as dt
import logging

from src.services.daily_tasks import remove_todo, set_todo_checked
from src.services.fast_lane.act import _find as find_event
from src.services.fast_lane.receipt import UNDO_WINDOW
from src.services.habit_completion import retract_completion

logger = logging.getLogger(__name__)

# The fields you can change in /day or by talking; a difference in any means "changed since".
_COMPARED = ("name", "start_time", "end_time", "state", "location", "activity", "proposed")


def _changed(current: dict, after: dict | None) -> bool:
    return after is None or any(current.get(k) != after.get(k) for k in _COMPARED)


def _retract(completion: dict, vault) -> str | None:
    result = retract_completion(vault, completion["path"], completion["completed_at"], completion["prior"])
    return None if result["ok"] else f"habit: {result['reason']}"


def _undo_created(rec: dict, *, events, vault, calendar) -> str | None:
    live = [e for e in (find_event(events, eid) for eid in rec["ids"]) if e is not None]
    if any(_changed(e, (rec.get("after") or {}).get(e["id"])) for e in live):
        return "changed since"
    for event in live:
        if event.get("habit_completion") and (why := _retract(event["habit_completion"], vault)):
            return why
        google_id = (event.get("source_ids") or {}).get("calendar_id")
        if google_id and calendar is not None and event.get("calendar") in (None, "Mazkir"):
            from src.services.async_bridge import maybe_await
            try:
                if not maybe_await(calendar.delete_event(google_id)):
                    return "Google refused to delete it"
            except Exception as e:
                return f"Google refused to delete it: {e}"
        events.delete_event(event["date"], event["id"])
    return None


def _undo_edited(rec: dict, *, events, vault, calendar) -> str | None:
    current = find_event(events, rec["event_id"])
    if current is None:
        return "the block is gone"
    before, after = rec.get("before") or {}, rec.get("after") or {}
    if _changed(current, after):
        return "changed since"
    ticked = current.get("habit_completion")
    if ticked and not before.get("habit_completion") and (why := _retract(ticked, vault)):
        return why
    keys = (set(before) | set(after)) - {"id", "date"}
    updates = {k: before.get(k) for k in keys if before.get(k) != after.get(k)}
    if updates:
        result = events.update_event(current["date"], rec["event_id"], updates)
        if "error" in result:
            return result["error"]
    if (current.get("source_ids") or {}).get("calendar_id") and calendar is not None:
        from src.services.calendar_push import sync_block
        restored = find_event(events, rec["event_id"])
        if restored is not None:
            sync_block(calendar, events, restored["date"], rec["event_id"])   # best effort; reported by its own log
    return None


def _undo_todo(rec: dict, *, vault, remove: bool) -> str | None:
    note = vault.read_daily_note(rec["date"])
    if remove:
        body, reason = remove_todo(note["content"], rec["text"])
        if reason == "not_found":
            return None                     # already gone: nothing of ours is left to take back
    else:
        body, reason = set_todo_checked(note["content"], rec["text"], checked=False)
        if reason == "not_found":
            return "the todo is gone"
    if reason != "ok":
        return f"several todos match “{rec['text']}”"
    vault.write_daily_note(rec["date"], body)
    return None


def _undo_one(rec: dict, *, events, vault, calendar) -> str | None:
    kind = rec["kind"]
    if kind == "created":
        return _undo_created(rec, events=events, vault=vault, calendar=calendar)
    if kind == "edited":
        return _undo_edited(rec, events=events, vault=vault, calendar=calendar)
    if kind in ("todo_added", "todo_checked"):
        return _undo_todo(rec, vault=vault, remove=kind == "todo_added")
    if kind == "habit":
        return _retract(rec, vault)
    return rec.get("why") or "this can't be undone"


def undo_turn(turn: dict, *, events, vault, calendar, now: dt.datetime) -> tuple[dict, str]:
    """Reverse every line that can be, newest first. Returns the updated turn and a toast."""
    created = dt.datetime.fromisoformat(turn["created_at"]).replace(tzinfo=None)
    if now.replace(tzinfo=None) - created >= UNDO_WINDOW:
        return turn, "Too late to undo — it has been a day"
    undone = kept = 0
    for line in reversed(turn["lines"]):
        if line["status"] not in ("done", "approved", "pending") or not line.get("undo"):
            continue
        why = None
        for rec in reversed(line["undo"]):
            try:
                why = _undo_one(rec, events=events, vault=vault, calendar=calendar)
            except Exception as e:   # one line's failure must not strand the rest
                logger.warning("fast lane undo failed on a line", exc_info=True)
                why = str(e)
            if why:
                break
        if why:
            line["undo_note"] = f"can't undo: {why}"
            kept += 1
        else:
            line["status"] = "undone"
            line.pop("undo_note", None)
            undone += 1
    return turn, f"Undone {undone}" + (f", {kept} kept" if kept else "")
```

```python
What `FAST_LANE_MODE=on` runs for every message, in a worker thread. It
never raises for anything a message can cause: a failed parse is route
"router_fallback" (the router takes the message, never the `mazkir`
fallback), and a failed write is a ✗ line on the receipt.

import logging
import time
from collections import Counter

@dataclass
class FastTurn:

    fallthrough_skill: str | None = None
    turn: dict | None = None                 # the saved record; None when nothing ran
    receipt: dict | None = None

def _safe(fn: Callable[[], Any], default: Any) -> Any:
    try:
        return fn()
    except Exception:
        logger.warning("fast lane step failed; continuing without it", exc_info=True)
        return default


def _near_days(now: dt.datetime) -> list[str]:
    return [(now.date() - dt.timedelta(days=1)).isoformat(), now.date().isoformat()]


def _later_days(now: dt.datetime) -> list[str]:
    return [(now.date() + dt.timedelta(days=k)).isoformat() for k in range(1, 8)]


def _targets(result: ParseResult, ctx: FastContext, events) -> dict[int, dict]:
    """The block each edit clause is about (spec §7.1's order, then the coming week by name)."""
    near = day_blocks(events, _near_days(ctx.now))
    selected = day_blocks(events, [ctx.selected_date]) if ctx.selected_date else []
    later: list[dict] | None = None   # read only when an edit asks for it
    out = {}
    for i, c in enumerate(result.clauses):
        if c.op in EDIT_OPS and c.target:
            if later is None:
                later = day_blocks(events, _later_days(ctx.now))
            block = find_target(c.target, reply_text=ctx.reply_to, selected=selected, today=near,
                                later=later, prefer_open=c.op == "end_block")
            if block is not None:
                out[i] = block
    return out


def _candidate_days(resolutions, now: dt.datetime) -> list[str]:
    """The days whose blocks a record can match: today, yesterday, and each placement's day and the one before."""
    days = {now.date(), now.date() - dt.timedelta(days=1)}
    for r in resolutions:
        p = r.placement if r is not None else None
        for moment in ((p.start, p.end) if p is not None else ()):
            if moment is not None:
                days |= {moment.date(), moment.date() - dt.timedelta(days=1)}
    return sorted(d.isoformat() for d in days)


def _suggestion_lines(records: list[dict], deps: FastDeps, habits: list[dict], now: dt.datetime) -> list[dict]:

        if name:

                "text": f"Track “{name}” as a habit?", "reason": "", "suggest": name,
                "start": None, "end": None, "end_precision": None, "day": None, "date": None, "event_id": None,
                "calendar_sync": None, "habit": None, "habit_candidate": None, "place_text": None,
                "pending": [], "undo": [],
            })
    return out


def _log_turn(ctx: FastContext, deps: FastDeps, lines: list[PlannedLine], records: list[dict],
              receipt: dict, *, save_memory: bool) -> None:
    """Spec §7.6: the turn is part of the agent's memory, as skill `fast-lane`."""
    tools = [
        {"name": s.tool, "params": {k: v for k, v in s.params.items() if not k.startswith("_")},
         "confidence": 1.0, "reasoning": "fast lane", "confirmed": True,
         "result_summary": {"status": rec["status"], "reason": rec["reason"]}}
        for line, rec in zip(lines, records) for s in line.steps
    ]
    emit_agent_turn({

        "pending_action_id": None, "prior_action_id": None, "iters": 0, "stop_reason": "fast_lane",
    })
    if save_memory:

def run_fast_turn(ctx: FastContext, deps: FastDeps, settings: ShadowSettings, *, message_id: int | None = None,
                  on_stage: Callable[[str], None] = lambda _stage: None, parse=None) -> FastTurn:
    parse = parse or parse_message
    repeat = _safe(lambda: act.find_repeat(deps.actions_dir, ctx.chat_id, message_id), None)
    if repeat is not None:

                         extra={"event_type": "fast_turn", "route": "router_fallback", "error": str(e)})
            span.set_attribute("mazkir.fast.route", "router_fallback")
            return FastTurn(route="router_fallback")
        parse_ms = round((time.monotonic() - started) * 1000)
        span.set_attribute("mazkir.fast.route", result.route)
        if result.fallthrough_skill:
            span.set_attribute("mazkir.fast.fallthrough_skill", result.fallthrough_skill)
        fast_count = sum(1 for c in result.clauses if c.op in FAST_OPS)
        if fast_count == 0:

            logger.info("fast_turn", extra={"event_type": "fast_turn", "route": "fallthrough",
                                            "clauses": len(result.clauses), "parse_ms": parse_ms})
            return FastTurn(route="fallthrough", fallthrough_skill=result.fallthrough_skill, lone_other=lone)

        on_stage(f"Placing {fast_count} {'thing' if fast_count == 1 else 'things'}…")

        with _tracer.start_as_current_span("fast.resolve"):

        inputs = PlanInputs(

            candidates=_safe(lambda: day_blocks(deps.events, _candidate_days(resolutions, ctx.now)), []),
            places=_safe(lambda: load_places(deps.places_path), []),
            todos=list(ctx.todos),
        )
        lines = plan_turn(result, resolutions, inputs)

        on_stage("Writing…")
        records = act.run_lines(lines, run_tool=deps.run_tool, events=deps.events)
        records += _suggestion_lines(records, deps, habits, ctx.now)
        turn_id = act.new_turn_id()

                "created_at": ctx.now.isoformat(), "text": ctx.text, "route": result.route,
                "fallthrough_skill": result.fallthrough_skill, "response": "", "lines": records}
        with _tracer.start_as_current_span("fast.receipt"):
            receipt = render_receipt(turn, ctx.now)
        _safe(lambda: (act.save_turn(deps.actions_dir, turn), act.prune(deps.actions_dir)), None)
        _safe(lambda: _log_turn(ctx, deps, lines, records, receipt, save_memory=result.route == "fast"), None)

        outcomes = Counter(rec["outcome"] for rec in records)
        span.set_attribute("mazkir.fast.clauses", len(result.clauses))
        for outcome in ("fact", "plan", "proposal", "question"):
            span.set_attribute(f"mazkir.fast.{outcome}s", outcomes.get(outcome, 0))
        logger.info("fast_turn", extra={"event_type": "fast_turn", "turn_id": turn_id, "route": result.route,
                                        "clauses": len(result.clauses), "outcomes": dict(outcomes),
                                        "parse_ms": parse_ms})
        return FastTurn(route=result.route, fallthrough_skill=result.fallthrough_skill, turn=turn,
                        receipt=receipt, ran=ran_summary(turn, ctx.now))
```

```python
"""Taps on a receipt (fast-lane spec §8.2): approve, dismiss, choose, add a place, confirm a habit, undo.

Each tap changes one line of the turn record and returns the re-rendered
receipt, which the bot edits in place. A line already settled answers with
a toast and changes nothing, so a double tap is harmless.
"""

from __future__ import annotations

import asyncio
import datetime as dt
from zoneinfo import ZoneInfo

from fastapi import APIRouter, Depends, HTTPException

from src.auth import verify_api_key
from src.config import settings
from src.services.fast_lane.act import apply_result, load_turn, run_step, save_turn

    """Module-level shim so tests can patch the services without importing src.main."""
    from src.main import get_agent, get_calendar, get_events, get_vault
    agent = get_agent()
    if agent is None:
        raise HTTPException(503, "Agent service not initialized")
    return {"run_tool": agent.run_tool, "events": get_events(), "vault": get_vault(),
            "calendar": get_calendar(), "data_dir": settings.events_data_path.parent}


def _now() -> dt.datetime:
    return dt.datetime.now(ZoneInfo(settings.vault_timezone))


def _turn(deps: dict, turn_id: str) -> dict:
    turn = load_turn(deps["data_dir"] / "actions", turn_id)
    if turn is None:
        raise HTTPException(404, "That receipt is gone — receipts are kept for a day")
    return turn


def _run(rec: dict, steps: list[dict], deps: dict) -> None:
    for s in steps:
        res = run_step(Step(s["tool"], s["params"]), run_tool=deps["run_tool"], events=deps["events"])
        apply_result(rec, res, done_status="done")
        if not res["ok"]:
            return


async def _set_state(rec: dict, state: str) -> None:
    from src.api.routes.events import SetStateBody, set_event_state
    await set_event_state(dt.date.fromisoformat(rec["date"]), rec["event_id"], SetStateBody(state=state))


async def _tap(action: str, turn: dict, rec: dict, deps: dict, now: dt.datetime) -> str:
    status = rec["status"]
    if action in ("ok", "no", "sy", "sn") and status != "pending":
        return "Already settled"
    if rec["kind"] == "suggest" and action in ("ok", "sy"):
        res = await asyncio.to_thread(deps["run_tool"], "create_habit", {"name": rec["suggest"], "frequency": "daily"})
        rec["status"] = "done" if res.get("ok") else "failed"
        return f"Tracking {rec['suggest']}" if res.get("ok") else "That didn't work"
    if rec["kind"] == "suggest" and action in ("no", "sn"):
        from src.services.fast_lane.habits import decline_suggestion
        decline_suggestion(rec["suggest"], now.date(), deps["data_dir"] / "fast-lane-suggestions.json")
        rec["status"] = "dismissed"
        return "Not now"
    if action == "ok":
        if rec["kind"] == "propose":
            await asyncio.to_thread(_run, rec, rec.get("pending") or [], deps)
            rec["pending"] = []
            return "Done" if rec["status"] == "done" else "That didn't work"
        await _set_state(rec, "approved")
        rec["status"] = "approved"
        event = next((e for e in deps["events"].get_events(rec["date"]) if e.get("id") == rec["event_id"]), {})
        rec["habit"] = event.get("habit_completion") or rec.get("habit")
        return "Approved"
    if action == "no":
        if rec["kind"] != "propose" and rec.get("event_id"):
            await _set_state(rec, "dismissed")
        rec["status"] = "dismissed"
        return "Dismissed"
    if action == "pl" and rec.get("place_text"):
        from src.services.fast_lane.places import add_place
        place = add_place(deps["data_dir"] / "places.json", rec["place_text"])
        if rec.get("event_id"):
            await asyncio.to_thread(deps["run_tool"], "update_event", {
                "event_id": rec["event_id"], "date": rec["date"],
                "_fast": {"defer_calendar_sync": True, "fields": {"place_id": place.id, "place_text": None}}})
        added, rec["place_text"] = rec["place_text"], None
        return f"Added {added}"
    if action == "hb" and rec.get("habit_candidate") and rec.get("event_id"):
        from src.services.fast_lane.habits import HabitMatch, tick_block_habit
        c = rec["habit_candidate"]
        rec["habit"] = await asyncio.to_thread(tick_block_habit, deps["vault"], deps["events"], rec["date"],
                                               rec["event_id"], HabitMatch(c["path"], c["name"], "fuzzy"))
        rec["habit_candidate"] = None
        return f"Ticked {c['name']}" if rec["habit"] else f"{c['name']} is already done for the day"
    raise HTTPException(404, "Nothing to do for that tap")


@router.get("/receipts/{turn_id}")
def get_receipt(turn_id: str):
    return render_receipt(_turn(get_receipt_deps(), turn_id), _now())


@router.post("/receipts/{turn_id}/{line}/{action}")
async def tap_receipt(turn_id: str, line: str, action: str):
    deps = get_receipt_deps()
    turn = _turn(deps, turn_id)
    now = _now()
    if action == "un":
        from src.services.fast_lane.undo import undo_turn
        turn, toast = await asyncio.to_thread(undo_turn, turn, events=deps["events"], vault=deps["vault"],
                                              calendar=deps["calendar"], now=now)
    else:
        try:
            rec = turn["lines"][int(line)]
        except (ValueError, IndexError):
            raise HTTPException(404, "No such line on that receipt")
        toast = await _tap(action, turn, rec, deps, now)
    save_turn(deps["data_dir"] / "actions", turn)
    return {"receipt": render_receipt(turn, now), "toast": toast}
```

```python
"""When a photo was taken, and the block it belongs to (fast-lane spec §5.6).

EXIF survives only when the image is sent as a file; a compressed Telegram
photo loses it (35 of 44 did). Then Telegram's send time is the best clue,
and it is UTC: read as local time it put photos three hours early, twice.
"""

from __future__ import annotations

import datetime as dt


def photo_moment(exif_timestamp: str | None, telegram_date: str | None, tz) -> tuple[dt.datetime | None, str | None]:
    """The moment, and "exact" (EXIF, local camera time) or "approx" (Telegram's send time, UTC)."""
    if exif_timestamp:
        try:
            return dt.datetime.fromisoformat(exif_timestamp).replace(tzinfo=tz), "exact"
        except ValueError:
            pass
    if telegram_date:
        try:
            sent = dt.datetime.fromisoformat(telegram_date.replace("Z", "+00:00"))
            if sent.tzinfo is None:
                sent = sent.replace(tzinfo=dt.timezone.utc)
            return sent.astimezone(tz).replace(microsecond=0), "approx"
        except ValueError:
            pass
    return None, None


def covering_block(events, moment: dt.datetime) -> dict | None:
    """The live block whose interval holds `moment`; an open block holds everything after its start.

    The latest-starting one wins when two overlap: it is the more specific.
    """
    day = moment.date().isoformat()
    wall = moment.replace(tzinfo=None)
    best = None
    for e in events.get_events(day):
        if e.get("state") == "dismissed" or not e.get("start_time"):
            continue
        start = dt.datetime.fromisoformat(e["start_time"]).replace(tzinfo=None)
        end = dt.datetime.fromisoformat(e["end_time"]).replace(tzinfo=None) if e.get("end_time") else None
        if start <= wall and (end is None or wall <= end) and (best is None or start > best[0]):
            best = (start, e)
    return {**best[1], "date": day} if best else None
```

```ts
import type { InputFile } from "grammy";
import type { InputRichMessage } from "@grammyjs/types";
import type { Receipt, ReceiptButton, ReceiptLine } from "@mazkir/shared-types";
import { escapeHtml } from "./telegram.js";

/** A fast-lane receipt (spec §8.2) as a rich message. The server decides every
 *  word, glyph and button; this only draws them, so a rule changes in one
 *  place. Buttons live in table cells: the one place Telegram renders them
 *  as compact pills rather than full-width bars (see day-rich.ts). */

function button(b: ReceiptButton): string {
  return `<tg-button type="callback_data" data="${escapeHtml(b.data)}">${escapeHtml(b.label)}</tg-button>`;
}

function cell(buttons: ReceiptButton[]): string {
  return buttons.length ? `<td><tg-button-row>${buttons.map(button).join("")}</tg-button-row></td>` : "<td></td>";
}

function row(line: ReceiptLine): string {
  const when = line.when ? ` ${escapeHtml(line.when)}` : "";
  const subs = [line.detail, line.sync, line.reason]
    .filter((s): s is string => Boolean(s))
    .map((s) => ` <sub>${escapeHtml(s)}</sub>`)
    .join("");
  return `<tr><td>${escapeHtml(line.glyph)}${when}</td><td>${escapeHtml(line.text)}${subs}</td>${cell(line.buttons)}</tr>`;
}

export function buildReceiptRich(receipt: Receipt): InputRichMessage<InputFile> {
  const rows = receipt.lines.map(row).join("");
  const undo = receipt.undo ? `<table><tr>${cell([{ label: "Undo", data: receipt.undo }])}</tr></table>` : "";
  return { html: `<table>${rows}</table>${undo}` };
}
```

```python
CACHE_MINIMUM_TOKENS = 4096   # Haiku 4.5 caches a prompt only from this size (fast-lane spec §5.3)
SLOW_PARSE_MS = 2500          # the top of §3.1's parse budget


def latency_summary(records: list[dict]) -> dict:
    """Parse times from shadow or fast-turn records: count, median and 95th percentile, in ms."""
    ms = sorted(r["parse_ms"] for r in records if isinstance(r.get("parse_ms"), int))
    if not ms:
        return {"n": 0, "p50": None, "p95": None}

    def at(q: float) -> int:
        return ms[min(len(ms) - 1, round(q * (len(ms) - 1)))]

    return {"n": len(ms), "p50": at(0.5), "p95": at(0.95)}


def caching_verdict(p50_ms: int | None, prompt_tokens: int) -> str:
    """Spec §13: grow the prompt past the cache minimum only if the parse is the slow part."""
    if prompt_tokens >= CACHE_MINIMUM_TOKENS:
        return "the prompt is already long enough to be cached"
    if p50_ms is None:
        return "no parses logged yet"
    if p50_ms > SLOW_PARSE_MS:
        return (f"parse p50 {p50_ms} ms is over budget: try growing the examples past "
                f"{CACHE_MINIMUM_TOKENS} tokens and measure again")
    return f"parse p50 {p50_ms} ms is within budget: leave the prompt as it is"
```

```python
"""How fast is the parse, and would prompt caching help? (fast-lane spec §5.3, §13)

    python scripts/fast_lane_latency.py            # from the shadow log; prompt size estimated
    python scripts/fast_lane_latency.py --count    # exact prompt tokens via count_tokens (free; needs the API key)
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))   # apps/vault-server

from src.config import settings  # noqa: E402
from src.services.fast_lane.parse import SYSTEM_PROMPT  # noqa: E402
from src.services.fast_lane.shadow_report import caching_verdict, latency_summary, load_jsonl  # noqa: E402


def prompt_tokens(count: bool) -> int:
    if not count:
        return round(len(SYSTEM_PROMPT) / 3.5)   # a rough estimate; --count asks the API
    import anthropic
    client = anthropic.Anthropic(api_key=settings.anthropic_api_key)
    return client.messages.count_tokens(model=settings.fast_parse_model, system=SYSTEM_PROMPT,
                                        messages=[{"role": "user", "content": "x"}]).input_tokens


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--count", action="store_true")
    args = ap.parse_args(argv)
    summary = latency_summary(load_jsonl(settings.logs_dir / "fast-lane-shadow.jsonl"))
    tokens = prompt_tokens(args.count)
    print(json.dumps({**summary, "prompt_tokens": tokens, "exact": args.count,
                      "verdict": caching_verdict(summary["p50"], tokens)}, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
